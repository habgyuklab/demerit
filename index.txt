<script>
    const groups = ['고1', '고2', '고3', 'N수', '성인'];
    const COLUMNS = 10;
    const ROWS = 3;
    const TOTAL_SEATS = COLUMNS * ROWS;
    const ITEMS_PER_PAGE = 30;
    let currentPage = 1;
    
    let appData = {
        members: {}, 
        penaltyReasons: [
            { id: 1, reason: '사유 지각', score: 1 },
            { id: 2, reason: '무단 지각', score: 2 },
            { id: 3, reason: '사유 결석', score: 2 },
            { id: 4, reason: '무단 결석', score: 10 },
            { id: 5, reason: '소음 발생', score: 3 },
            { id: 6, reason: '학습태도 불량', score: 5 },
            { id: 7, reason: '휴대폰 무단사용', score: 5 }
        ]
    };
    
    let currentSelectedId = null; 
    let editingRuleId = null; 

    // 초기화
    function initSeats() {
        const container = document.getElementById('main-container');
        container.innerHTML = '';
        
        groups.forEach(group => {
            const rowDiv = document.createElement('div');
            rowDiv.className = 'group-row';
            
            const labelDiv = document.createElement('div');
            labelDiv.className = 'group-label';
            labelDiv.innerText = group;
            rowDiv.appendChild(labelDiv);

            const gridDiv = document.createElement('div');
            gridDiv.className = 'grid-container';

            for(let i=0; i<TOTAL_SEATS; i++) {
                const seatId = `${group}_${i}`;
                if(!appData.members[seatId]) {
                    appData.members[seatId] = { name: '', penalties: [] };
                }

                const seatDiv = document.createElement('div');
                seatDiv.className = 'seat';
                seatDiv.id = seatId;
                seatDiv.addEventListener('click', (e) => showContextMenu(e, seatId));
                
                gridDiv.appendChild(seatDiv);
            }
            rowDiv.appendChild(gridDiv);
            container.appendChild(rowDiv);
        });
        updateAllSeats();
        updateRanking();
    }

    function getTotalScore(seatId) {
        return appData.members[seatId].penalties.reduce((sum, p) => sum + p.score, 0);
    }

    function getStatusAndColor(score) {
        if(score <= 10) return { status: '안정', color: 'var(--green)' };
        if(score >= 11 && score <= 20) return { status: '주의', color: 'var(--yellow)' };
        if(score >= 21 && score <= 29) return { status: '위험', color: 'var(--orange)' };
        return { status: '퇴실', color: 'var(--red)' };
    }

    function updateAllSeats() {
        for(let id in appData.members) {
            const seat = document.getElementById(id);
            if(!seat) continue;
            if(appData.members[id].name) {
                seat.innerText = appData.members[id].name;
                const score = getTotalScore(id);
                seat.style.backgroundColor = getStatusAndColor(score).color;
                seat.style.color = (score >= 30) ? 'white' : 'black';
            } else {
                seat.innerText = '';
                seat.style.backgroundColor = 'var(--gray)';
            }
        }
    }

    // 컨텍스트 메뉴
    const contextMenu = document.getElementById('context-menu');
    
    function showContextMenu(e, seatId) {
        currentSelectedId = seatId;
        contextMenu.style.display = 'block';
        
        let x = e.pageX;
        let y = e.pageY;
        if(x + 150 > window.innerWidth) x -= 150;
        if(y + 120 > window.innerHeight) y -= 120;
        
        contextMenu.style.left = `${x}px`;
        contextMenu.style.top = `${y}px`;
    }

    document.addEventListener('click', (e) => {
        if (!e.target.classList.contains('seat')) {
            contextMenu.style.display = 'none';
        }
    });

    // 이름 변경
    document.getElementById('menu-name').addEventListener('click', () => {
        const currentName = appData.members[currentSelectedId].name;
        const newName = prompt('회원 이름을 입력하세요:', currentName);
        if(newName !== null) {
            appData.members[currentSelectedId].name = newName.trim();
            updateAllSeats();
            updateRanking();
        }
    });

    // 벌점 조항 관리
    function openRuleModal() {
        renderRules();
        document.getElementById('modal-rules').style.display = 'flex';
    }

    function renderRules() {
        const container = document.getElementById('rule-list-container');
        container.innerHTML = '';
        appData.penaltyReasons.forEach(rule => {
            const div = document.createElement('div');
            div.className = 'flex-row';
            div.innerHTML = `
                <span style="flex:1; font-size:15px;"><b>${rule.reason}</b> (${rule.score}점)</span>
                <div>
                    <button class="btn-sm btn-edit" onclick="openRuleForm(${rule.id})">수정</button>
                    <button class="btn-sm btn-delete" onclick="deleteRule(${rule.id})">삭제</button>
                </div>
            `;
            container.appendChild(div);
        });
    }

    function openRuleForm(id) {
        editingRuleId = id;
        const reasonInput = document.getElementById('input-rule-reason');
        const scoreInput = document.getElementById('input-rule-score');
        
        if (id === null) {
            document.getElementById('rule-form-title').innerText = '새 벌점 조항 추가';
            reasonInput.value = '';
            scoreInput.value = '';
        } else {
            document.getElementById('rule-form-title').innerText = '벌점 조항 수정';
            const rule = appData.penaltyReasons.find(r => r.id === id);
            reasonInput.value = rule.reason;
            scoreInput.value = rule.score;
        }
        
        document.getElementById('modal-rule-form').style.display = 'flex';
        reasonInput.focus();
    }

    function saveRule() {
        const reason = document.getElementById('input-rule-reason').value.trim();
        const scoreStr = document.getElementById('input-rule-score').value.trim();
        const score = parseInt(scoreStr, 10);

        if (!reason) return alert('벌점 사유를 입력해주세요.');
        if (!scoreStr || isNaN(score)) return alert('벌점 점수를 정확히 입력해주세요.');

        if (editingRuleId === null) {
            appData.penaltyReasons.push({ id: Date.now(), reason, score });
        } else {
            const rule = appData.penaltyReasons.find(r => r.id === editingRuleId);
            if (rule) {
                rule.reason = reason;
                rule.score = score;
            }
        }
        renderRules();
        closeModal('modal-rule-form');
    }

    function deleteRule(id) {
        if(confirm('해당 벌점 조항을 삭제하시겠습니까?\n(이미 부과된 회원의 기록은 사라지지 않습니다)')) {
            appData.penaltyReasons = appData.penaltyReasons.filter(r => r.id !== id);
            renderRules();
        }
    }

    // 벌점 부과
    document.getElementById('menu-add-penalty').addEventListener('click', () => {
        if(!appData.members[currentSelectedId].name) return alert('먼저 이름을 등록해주세요.');
        
        const container = document.getElementById('penalty-options-container');
        container.innerHTML = '';
        
        appData.penaltyReasons.forEach(p => {
            const div = document.createElement('div');
            div.className = 'penalty-option';
            div.innerHTML = `<span style="font-weight:bold;">${p.reason}</span> <span>${p.score}점</span>`;
            div.onclick = () => applyPenalty(p.reason, p.score);
            container.appendChild(div);
        });
        
        document.getElementById('modal-penalty').style.display = 'flex';
    });

    function applyPenalty(reason, score) {
        const now = new Date();
        const m = now.getMonth() + 1;
        const d = now.getDate();
        const h = String(now.getHours()).padStart(2, '0');
        const min = String(now.getMinutes()).padStart(2, '0');
        const sec = String(now.getSeconds()).padStart(2, '0');
        const dateStr = `${m}/${d} ${h}:${min}:${sec}`; 
        
        appData.members[currentSelectedId].penalties.push({
            id: Date.now(),
            date: dateStr,
            reason: reason,
            score: score
        });
        
        closeModal('modal-penalty');
        updateAllSeats();
        updateRanking();
        
        // 벌점 부과 팝업(alert) 제거됨
    }

    // 메뉴 -> 벌점 현황
    document.getElementById('menu-history').addEventListener('click', () => {
        if(!appData.members[currentSelectedId].name) return alert('먼저 이름을 등록해주세요.');
        openHistoryModal(currentSelectedId);
    });

    // 순위표 내역 버튼 클릭 시 바로 팝업 오픈
    function openHistoryFromRank(seatId) {
        openHistoryModal(seatId);
    }

    function openHistoryModal(seatId) {
        currentSelectedId = seatId; // 현재 선택된 학생 ID 세팅
        const member = appData.members[currentSelectedId];
        document.getElementById('history-title').innerText = `${member.name} 벌점 현황`;
        renderHistory();
        document.getElementById('modal-history').style.display = 'flex';
    }

    function renderHistory() {
        const listContainer = document.getElementById('history-list');
        listContainer.innerHTML = '';
        const member = appData.members[currentSelectedId];
        
        if(member.penalties.length === 0) {
            listContainer.innerHTML = '<p style="text-align:center; color:#7f8c8d; padding: 20px;">부과된 벌점 내역이 없습니다.</p>';
            return;
        }

        let runningTotal = 0;
        member.penalties.forEach(p => {
            runningTotal += p.score;
            const div = document.createElement('div');
            div.className = 'flex-row';
            div.style.fontSize = '13px';
            div.innerHTML = `
                <span style="width: 110px; color:#555;">${p.date}</span>
                <span style="flex: 1; font-weight:bold;">${p.reason}</span>
                <span style="width: 40px; text-align:right;">${p.score}점</span>
                <span style="width: 70px; text-align:right; font-weight:bold; color:var(--red);">총 ${runningTotal}점</span>
                <button class="btn-sm btn-delete" style="margin-left:15px;" onclick="deletePenalty(${p.id})">삭제</button>
            `;
            listContainer.appendChild(div);
        });
    }

    function deletePenalty(penaltyId) {
        if(confirm('이 벌점 내역을 삭제하시겠습니까?')) {
            appData.members[currentSelectedId].penalties = appData.members[currentSelectedId].penalties.filter(p => p.id !== penaltyId);
            renderHistory();
            updateAllSeats();
            updateRanking();
        }
    }

    function closeModal(id) { document.getElementById(id).style.display = 'none'; }

    // 순위 및 페이지네이션 (공동 등수 로직 적용)
    function updateRanking() {
        const listContainer = document.getElementById('ranking-list');
        listContainer.innerHTML = '';
        
        const currentMonth = new Date().getMonth() + 1;
        document.getElementById('ranking-title').innerText = `${currentMonth}월 벌점 순위`;

        let rankingData = [];
        for(let id in appData.members) {
            if(appData.members[id].name) {
                const score = getTotalScore(id);
                rankingData.push({ 
                    id: id, 
                    name: appData.members[id].name, 
                    score: score, 
                    statusInfo: getStatusAndColor(score) 
                });
            }
        }
        
        // 내림차순 정렬
        rankingData.sort((a, b) => b.score - a.score);

        // 공동 순위 계산
        let currentRank = 1;
        for (let i = 0; i < rankingData.length; i++) {
            if (i > 0 && rankingData[i].score < rankingData[i - 1].score) {
                currentRank = i + 1;
            }
            rankingData[i].rank = currentRank;
        }

        const totalPages = Math.ceil(rankingData.length / ITEMS_PER_PAGE) || 1;
        if(currentPage > totalPages) currentPage = totalPages;

        const startIndex = (currentPage - 1) * ITEMS_PER_PAGE;
        const pageData = rankingData.slice(startIndex, startIndex + ITEMS_PER_PAGE);

        pageData.forEach((data) => {
            const item = document.createElement('div');
            item.className = 'ranking-item';
            item.innerHTML = `
                <span class="rank">${data.rank}등</span>
                <span class="name">${data.name}</span>
                <span class="status" style="color: ${data.statusInfo.color}">${data.statusInfo.status}</span>
                <span class="score">${data.score}점</span>
                <button class="btn-history-view" onclick="openHistoryFromRank('${data.id}')">내역</button>
            `;
            listContainer.appendChild(item);
        });

        const pagination = document.getElementById('pagination');
        pagination.innerHTML = '';
        for(let i=1; i<=totalPages; i++) {
            const btn = document.createElement('button');
            btn.className = `page-btn ${i === currentPage ? 'active' : ''}`;
            btn.innerText = i;
            btn.onclick = () => { currentPage = i; updateRanking(); };
            pagination.appendChild(btn);
        }
        window.currentRankingData = rankingData; 
    }

    // 텍스트 다운로드 (공동 등수 적용)
    document.getElementById('btn-download').addEventListener('click', () => {
        const currentMonth = new Date().getMonth() + 1;
        let textContent = `[ ${currentMonth}월 전회원 벌점 순위 ]\n-------------------------------------------------\n등수\t이름\t상태\t누적벌점\n-------------------------------------------------\n`;
        if(!window.currentRankingData || window.currentRankingData.length === 0) {
            textContent += `등록된 회원이 없습니다.\n`;
        } else {
            window.currentRankingData.forEach((data) => {
                textContent += `${data.rank}등\t${data.name}\t${data.statusInfo.status}\t${data.score}점\n`;
            });
        }
        downloadFile(textContent, `${currentMonth}월_전회원_벌점순위.txt`, 'text/plain');
    });

    // JSON 다운/불러오기 
    function downloadJSON() {
        const jsonStr = JSON.stringify(appData, null, 2);
        
        const now = new Date();
        const month = now.getMonth() + 1;
        const day = now.getDate();
        
        const fileName = `합격랩 ${month}월 ${day}일 벌점현황.json`;
        
        downloadFile(jsonStr, fileName, 'application/json');
    }
    
    document.getElementById('btn-download-json').addEventListener('click', downloadJSON);

    document.getElementById('file-load-json').addEventListener('change', (e) => {
        const file = e.target.files[0];
        if(!file) return;
        const reader = new FileReader();
        reader.onload = function(event) {
            try {
                const parsed = JSON.parse(event.target.result);
                if(parsed.members && parsed.penaltyReasons) {
                    appData = parsed;
                    initSeats();
                    alert('데이터를 성공적으로 불러왔습니다!');
                } else {
                    alert('올바른 백업 파일이 아닙니다.');
                }
            } catch(e) {
                alert('파일을 읽는 중 오류가 발생했습니다.');
            }
        };
        reader.readAsText(file);
        e.target.value = ''; 
    });

    function downloadFile(content, fileName, mimeType) {
        const blob = new Blob([content], { type: mimeType + ';charset=utf-8' });
        const url = URL.createObjectURL(blob);
        const a = document.createElement('a');
        a.href = url; a.download = fileName;
        document.body.appendChild(a);
        a.click();
        document.body.removeChild(a);
        URL.revokeObjectURL(url);
    }

    // 자동 백업 타이머
    let hasAutoDownloadedToday = false;
    setInterval(() => {
        const now = new Date();
        const hour = now.getHours();
        const minute = now.getMinutes();

        if (hour === 18 && minute === 0 && !hasAutoDownloadedToday) {
            downloadJSON();
            hasAutoDownloadedToday = true;
            console.log("자동 백업이 완료되었습니다.");
        }
        
        if (hour === 0) {
            hasAutoDownloadedToday = false;
        }
    }, 30000);
